class IDMapper:

    def get(self, key):
        if key not in self.map: self.map[key] = len(self.map)
        return self.map[key]

# =========================================================
def _bool_attr(attrs, key):
    if not attrs: return 0.0
    v = attrs.get(key)
    if v is None: return 0.0
    if isinstance(v, bool): return float(v)
    if isinstance(v, str):
        return 1.0 if v.strip("'\" ").lower() in ("true","yes","free","paid","1") else 0.0
    return 0.0

def _parse_total_hours(hours):
    if not hours: return 0, 0.0
    open_days, total = len(hours), 0.0
    for _, hrs in hours.items():
        if not hrs or hrs == "0:0-0:0": continue
        try:
            o, c = hrs.split("-")
            oh, om = map(int, o.split(":"))
            ch, cm = map(int, c.split(":"))
            total += max(0, (ch*60+cm)-(oh*60+om)) / 60.0
        except: total += 8.0
    return open_days, total

def _parse_nested_bool(attrs, outer_key, inner_key):
    if not attrs: return 0.0
    raw = attrs.get(outer_key)
    if raw is None: return 0.0
    if isinstance(raw, dict):
        return 1.0 if str(raw.get(inner_key,"")).lower() in ("true","1","yes") else 0.0
    if isinstance(raw, str):
        return 1.0 if f"'{inner_key}': True" in raw else 0.0
    return 0.0

def build_city_map(path, top_n=50):
    counter = Counter()
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            b = json.loads(line)
            counter[(b.get("city") or "Unknown").strip()] += 1
    city_map = {c:(i+1) for i,(c,_) in enumerate(counter.most_common(top_n))}
    print(f"  ✓ City map: top {top_n} cities"); return city_map

    state_map = {s:(i+1) for i,(s,_) in enumerate(counter.most_common(top_n))}
    print(f"  ✓ State map: top {top_n} states"); return state_map

# BUILD SOCIAL GROUP MAP (KMeans k=5)
# =========================================================
def build_social_group_map(user_path, n_clusters=N_SOCIAL_GROUPS):
    print(f"  Building social groups (k={n_clusters})...")
    users = []
    with open(user_path, "r", encoding="utf-8") as f:
        for line in f:
            u = json.loads(line)
            elite = u.get("elite", "")
            elite_years = len([e for e in elite.split(",") if e.strip()]) if isinstance(elite, str) else 0
            friends = u.get("friends", "")
            friend_count = len([x for x in friends.split(",") if x.strip() and x != "None"])
            users.append({
                "user_id":      u["user_id"],
                "review_count": u.get("review_count", 0),
                "elite_years":  elite_years,
                "friend_count": friend_count,
            })

    import pandas as pd
    df = pd.DataFrame(users)
    X = StandardScaler().fit_transform(df[["review_count", "elite_years", "friend_count"]])

    sample_idx = np.random.choice(len(X), size=min(50_000, len(X)), replace=False)
    km = KMeans(n_clusters=n_clusters, random_state=42, n_init=10).fit(X[sample_idx])
    df["group_id"] = km.predict(X)

    group_map = dict(zip(df["user_id"], df["group_id"]))
    print(f"  ✓ Social groups: {len(group_map):,} users → {n_clusters} groups")
    return group_map

# LOAD KG EMBEDDINGS
# =========================================================
def load_kg_embeddings(kg_emb_dir, entity_mapping_dir):
    print("  Loading KG embeddings...")
    emb = np.load(f"{kg_emb_dir}/entity_embeddings.npy")  # (n_entities, 64)

    with open(f"{kg_emb_dir}/pykeen_entity_to_id.json") as f:
        pykeen_map = json.load(f)  # {"global_id_str": pykeen_id}

    with open(f"{entity_mapping_dir}/entity_mapping.json") as f:
        entity_map = json.load(f)  # {"U_xxx": global_id, "B_yyy": global_id}

    user_kg_emb = {}
    biz_kg_emb  = {}
    for key, global_id in entity_map.items():
        pykeen_id = pykeen_map.get(str(global_id))
        if pykeen_id is None:
            continue
        vec = emb[pykeen_id]
        if key.startswith("U_"):
            user_kg_emb[key[2:]] = vec.real
        elif key.startswith("B_"):
            biz_kg_emb[key[2:]]  = vec.real

    print(f"  ✓ KG embeddings: {len(user_kg_emb):,} users, {len(biz_kg_emb):,} businesses")
    return user_kg_emb, biz_kg_emb

# LOAD USER CONTEXT — 2 features

def load_user_context(path):
    ctx = {}

    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            u = json.loads(line)
            uid = u["user_id"]

            try:
                dt = datetime.strptime(
                    u.get("yelping_since", "2015-01-01 00:00:00"),
                    "%Y-%m-%d %H:%M:%S"
                )
                years_active = max(0.0, (2024 - dt.year) / 10.0)
            except:
                years_active = 0.5

            ctx[uid] = [
                math.log1p(u.get("review_count", 0)) / 10.0,
                years_active,
            ]

    print(f"  ✓ Users: {len(ctx):,}")
    return ctx

# LOAD CHECKIN CONTEXT — 4 features
# =========================================================
def load_checkin_context(path):
    ctx = {}
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            c = json.loads(line)
            bid = c["business_id"]
            dates_str = c.get("date","")
            if not dates_str:
                ctx[bid] = [0.0,0.0,0.0,0.0]; continue

            timestamps = [d.strip() for d in dates_str.split(",") if d.strip()]
            count = len(timestamps)
            weekend_count, peak_hours, parsed_dts = 0, [], []
            for ts in timestamps:
                try:
                    dt = datetime.strptime(ts,"%Y-%m-%d %H:%M:%S")
                    parsed_dts.append(dt)
                    if dt.weekday() >= 5: weekend_count += 1
                    peak_hours.append(dt.hour)
                except: pass

            weekend_ratio = weekend_count/count if count > 0 else 0.0
            peak_hour = Counter(peak_hours).most_common(1)[0][0]/23.0 if peak_hours else 0.0

            if len(parsed_dts) >= 2:
                span_weeks = max((max(parsed_dts) - min(parsed_dts)).days / 7.0, 1.0)
                density = count / span_weeks
            else:
                density = float(count)

            ctx[bid] = [math.log1p(count)/10.0, math.log1p(density)/10.0, weekend_ratio, peak_hour]
    print(f"  ✓ Checkin: {len(ctx):,} businesses"); return ctx

# LOAD TIP CONTEXT

def load_tip_context(path):
    user_tip, biz_tip = {}, {}
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            t = json.loads(line)
            uid, bid = t["user_id"], t["business_id"]
            total_len = len(t.get("text",""))
            if uid not in user_tip: user_tip[uid] = [0,0]
            user_tip[uid][0]+=1; user_tip[uid][1]+=total_len
            if bid not in biz_tip:  biz_tip[bid]  = [0,0]
            biz_tip[bid][0]+=1;  biz_tip[bid][1]+=total_len
    print(f"  ✓ Tips: {len(user_tip):,} users, {len(biz_tip):,} businesses")
    return user_tip, biz_tip


# =========================================================
# LOAD ALL DATA — KG tách riêng khỏi num_arr
# =========================================================
def load_all_data(review_path, user_ctx, biz_ctx, user_tip, biz_tip, checkin_ctx,
                  user_kg_emb=None, biz_kg_emb=None,
                  max_reviews=None, cold_threshold=COLD_START_THRESHOLD,
                  cache_dir="/kaggle/working", selected_features=None):
    import pandas as pd
    # Cache hash thêm "_v2" để tránh load nhầm cache cũ (195-dim)
    cache_tag  = f"{FEAT_HASH}_v2"
    cache_file = os.path.join(cache_dir, f"dataset_cache_{cache_tag}.npz")
    map_file   = os.path.join(cache_dir, f"dataset_cache_{cache_tag}_maps.pkl")
    kg_file    = os.path.join(cache_dir, f"dataset_cache_{cache_tag}_kg.npz")

    def filter_features(arr):
        if selected_features is not None:
            idx = [FEAT_NAMES.index(f) for f in selected_features if f in FEAT_NAMES]
            if not idx:
                print("  INFO: 0 features selected! Running as pure Matrix Factorization + KG.")
                return arr[:, []]
            print(f"  Filtering → {len(idx)} features selected")
            return arr[:, idx]
        return arr

    # --- TRY LOAD CACHE ---
    if max_reviews is None and os.path.exists(cache_file) and os.path.exists(map_file):
        print(f"  Loading from cache ({cache_tag})...")
        try:
            d = np.load(cache_file)
            all_data = TensorDataset(
                torch.from_numpy(d["users"]).long(), torch.from_numpy(d["bizs"]).long(),
                torch.from_numpy(filter_features(d["num"])),
                torch.from_numpy(d["ratings"]), torch.from_numpy(d["coldu"]), torch.from_numpy(d["coldb"]))
            with open(map_file,"rb") as f: maps = pickle.load(f)

            # Load KG matrices từ file riêng
            user_kg_mat, biz_kg_mat = None, None
            if os.path.exists(kg_file):
                kg_data = np.load(kg_file)
                user_kg_mat = kg_data.get("user_kg_mat")
                biz_kg_mat  = kg_data.get("biz_kg_mat")
                if user_kg_mat is not None:
                    print(f"  ✓ KG cache loaded: user={user_kg_mat.shape}, biz={biz_kg_mat.shape}")
                else:
                    print("  ⚠️  KG cache file exists but no data inside, will be None")
            else:
                print("  ⚠️  No KG cache found, KG embeddings will be None")

            print(f"  Cache loaded: {len(all_data):,} samples")
            return all_data, maps["user_map"], maps["biz_map"], maps["user_counts"], maps["biz_counts"], user_kg_mat, biz_kg_mat
        except Exception as e:
            print(f"  Cache failed: {e}. Rebuilding...")

    # --- BUILD FROM SCRATCH ---
    print("  [Single Pass] Reading reviews...")
    user_counts, biz_counts = Counter(), Counter()
    user_map, biz_map = IDMapper(), IDMapper()
    uid_l,bid_l,stars_l,date_l = [],[],[],[]

    with open(review_path,"rb") as f:
        for i, line in enumerate(f):
            if max_reviews and i >= max_reviews: break
            r = orjson.loads(line)
            uid, bid = r["user_id"], r["business_id"]
            user_counts[uid]+=1; biz_counts[bid]+=1
            uid_l.append(user_map.get(uid)); bid_l.append(biz_map.get(bid))
            stars_l.append(r["stars"]); date_l.append(r["date"])

    N = len(uid_l); n_users = len(user_map.map); n_bizs = len(biz_map.map)
    print(f"  {N:,} reviews | {n_users:,} users | {n_bizs:,} businesses")

    users_arr = np.array(uid_l, dtype=np.int64)
    bizs_arr  = np.array(bid_l, dtype=np.int64)

    user_mat = np.zeros((n_users, USER_CTX_DIM), dtype=np.float32)
    for uid, feats in user_ctx.items():
        if uid in user_map.map: user_mat[user_map.map[uid]] = feats

    biz_mat = np.zeros((n_bizs, BIZ_CTX_DIM), dtype=np.float32)
    for bid, feats in biz_ctx.items():
        if bid in biz_map.map: biz_mat[biz_map.map[bid]] = feats

    utip_mat = np.zeros((n_users, 2), dtype=np.float32)
    for uid, t in user_tip.items():
        if uid in user_map.map:
            cnt = max(t[0],1)
            utip_mat[user_map.map[uid]] = [math.log1p(t[0])/10, math.log1p(t[1]/cnt)/10]

    btip_mat = np.zeros((n_bizs, 2), dtype=np.float32)
    for bid, t in biz_tip.items():
        if bid in biz_map.map:
            cnt = max(t[0],1)
            btip_mat[biz_map.map[bid]] = [math.log1p(t[0])/10, math.log1p(t[1]/cnt)/10]

    ck_mat = np.zeros((n_bizs, CHECKIN_DIM), dtype=np.float32)
    for bid, c in checkin_ctx.items():
        if bid in biz_map.map: ck_mat[biz_map.map[bid]] = c

    dts = pd.to_datetime(date_l)
    temporal_feat = np.stack([
        (dts.year-2015).values/10.0, dts.month.values/12.0,
        dts.day.values/31.0, dts.weekday.values/7.0, dts.hour.values/24.0,
    ], axis=1).astype(np.float32)

    # --- KG MATRICES (tách riêng, KHÔNG concat vào num_arr) ---
    user_kg_mat = None
    biz_kg_mat  = None

    if user_kg_emb:
        user_kg_mat = np.zeros((n_users, KG_DIM), dtype=np.float32)
        matched_users = 0
        for uid, vec in user_kg_emb.items():
            if uid in user_map.map:
                user_kg_mat[user_map.map[uid]] = vec
                matched_users += 1
        coverage_u = matched_users / max(n_users, 1) * 100
        print(f"  ✓ User KG matrix: {user_kg_mat.shape} | Coverage: {matched_users:,}/{n_users:,} ({coverage_u:.1f}%)")

    if biz_kg_emb:
        biz_kg_mat = np.zeros((n_bizs, KG_DIM), dtype=np.float32)
        matched_bizs = 0
        for bid, vec in biz_kg_emb.items():
            if bid in biz_map.map:
                biz_kg_mat[biz_map.map[bid]] = vec
                matched_bizs += 1
        coverage_b = matched_bizs / max(n_bizs, 1) * 100
        print(f"  ✓ Biz KG matrix:  {biz_kg_mat.shape} | Coverage: {matched_bizs:,}/{n_bizs:,} ({coverage_b:.1f}%)")

    # --- CONCAT CHỈ 67 CONTEXT FEATURES ---
    num_arr = np.concatenate([
        user_mat[users_arr],
        biz_mat[bizs_arr],
        utip_mat[users_arr],
        btip_mat[bizs_arr],
        ck_mat[bizs_arr],
        temporal_feat,
    ], axis=1)
    assert num_arr.shape[1] == NUM_DIM, f"Shape mismatch: {num_arr.shape[1]} != {NUM_DIM}"

    u_cnts = np.bincount(users_arr, minlength=n_users)
    b_cnts = np.bincount(bizs_arr,  minlength=n_bizs)
    coldu_arr = (u_cnts[users_arr] <= cold_threshold).astype(np.float32)
    coldb_arr = (b_cnts[bizs_arr]  <= cold_threshold).astype(np.float32)
    ratings_arr = ((np.array(stars_l, dtype=np.float32) - TARGET_MIN) / TARGET_RANGE)

    filtered = filter_features(num_arr)
    all_data = TensorDataset(
        torch.from_numpy(users_arr).long(), torch.from_numpy(bizs_arr).long(),
        torch.from_numpy(filtered), torch.from_numpy(ratings_arr),
        torch.from_numpy(coldu_arr), torch.from_numpy(coldb_arr))

    # --- SAVE CACHE ---
    if max_reviews is None:
        os.makedirs(cache_dir, exist_ok=True)
        print(f"  Caching to {cache_file}...")
        np.savez_compressed(cache_file, users=users_arr, bizs=bizs_arr,
                            num=num_arr, ratings=ratings_arr, coldu=coldu_arr, coldb=coldb_arr)
        with open(map_file,"wb") as f:
            pickle.dump({"user_map":user_map,"biz_map":biz_map,
                         "user_counts":user_counts,"biz_counts":biz_counts}, f)
        # KG lưu riêng file để tránh pickle numpy array lớn
        kg_save = {}
        if user_kg_mat is not None: kg_save["user_kg_mat"] = user_kg_mat
        if biz_kg_mat  is not None: kg_save["biz_kg_mat"]  = biz_kg_mat
        if kg_save:
            np.savez_compressed(kg_file, **kg_save)
            print(f"  ✓ KG matrices cached → {kg_file}")

    return all_data, user_map, biz_map, user_counts, biz_counts, user_kg_mat, biz_kg_mat


# =========================================================
# CROSS&COMPRESS UNIT — Toggle between Matrix and Vector
# =========================================================
class CrossCompressUnit(nn.Module):
    def __init__(self, dim: int, use_matrix: bool = True):
        """
        use_matrix = True: Generalized CC Unit (Matrix weights, Full-rank) ~33k params
        use_matrix = False: Faithful MKR CC Unit (Vector weights, Low-rank) ~384 params
        """
        super().__init__()
        self.dim = dim
        self.use_matrix = use_matrix

        if self.use_matrix:
            # Generalized version (Full-rank matrix weights)
            self.W_VV = nn.Linear(dim, dim, bias=False)
            self.W_EV = nn.Linear(dim, dim, bias=False)
            self.W_VE = nn.Linear(dim, dim, bias=False)
            self.W_EE = nn.Linear(dim, dim, bias=False)
            
            for w in [self.W_VV.weight, self.W_EV.weight, self.W_VE.weight, self.W_EE.weight]:
                nn.init.normal_(w, 0, 0.01)
        else:
            # Paper version (MKR faithful vector weights)
            self.w_VV = nn.Parameter(torch.zeros(dim))
            self.w_EV = nn.Parameter(torch.zeros(dim))
            self.w_VE = nn.Parameter(torch.zeros(dim))
            self.w_EE = nn.Parameter(torch.zeros(dim))
            
            for w in [self.w_VV, self.w_EV, self.w_VE, self.w_EE]:
                nn.init.normal_(w, 0, 0.01)
                
        self.b_V = nn.Parameter(torch.zeros(dim))
        self.b_E = nn.Parameter(torch.zeros(dim))

    def forward(self, v_item: torch.Tensor, v_entity: torch.Tensor):
        # Outer product: C[b,i,j] = v_item[b,i] * v_entity[b,j]
        # (B, d, 1) × (B, 1, d) → (B, d, d)
        C   = torch.bmm(v_item.unsqueeze(2), v_entity.unsqueeze(1))
        C_T = C.transpose(1, 2)
        
        if self.use_matrix:
            # nn.Linear applied on (B, d, d) acts on the last dimension
            # .sum(dim=1) aggregates over the rows, resulting in (B, d)
            v_item_out = self.W_VV(C).sum(dim=1) + self.W_EV(C_T).sum(dim=1) + self.b_V
            v_ent_out  = self.W_VE(C).sum(dim=1) + self.W_EE(C_T).sum(dim=1) + self.b_E
        else:
            # C · w: (B, d, d) × (d,) → (B, d)
            v_item_out = (torch.matmul(C,   self.w_VV) +
                          torch.matmul(C_T, self.w_EV) + self.b_V)
            v_ent_out  = (torch.matmul(C,   self.w_VE) +
                          torch.matmul(C_T, self.w_EE) + self.b_E)
            
        return v_item_out, v_ent_out


# =========================================================
# FM MODEL — KG Projection + Cross&Compress + Zero-Mask
# =========================================================
class FMModel(nn.Module):
    def __init__(self, n_user, n_business, num_dim, k=64, dropout=0.15,
                 user_kg_mat=None, biz_kg_mat=None, use_matrix_cc=True):
        super().__init__()
        self.dropout = nn.Dropout(dropout)
        self.k = k
        self.num_dim = num_dim

        # --- BIAS ---
        self.bias = nn.Parameter(torch.tensor([0.5]))

        # --- FIRST-ORDER (linear terms) ---
        self.user_lin = nn.Embedding(n_user, 1)
        self.biz_lin  = nn.Embedding(n_business, 1)
        self.num_lin  = nn.Parameter(torch.zeros(num_dim)) if num_dim > 0 else None

        # --- SECOND-ORDER (embedding terms cho FM interaction) ---
        self.user_emb = nn.Embedding(n_user, k)
        self.biz_emb  = nn.Embedding(n_business, k)
        self.num_emb  = nn.Parameter(torch.zeros(num_dim, k)) if num_dim > 0 else None

        # --- COLD-START FALLBACK ---
        self.avg_user_emb = nn.Parameter(torch.zeros(k))
        self.avg_biz_emb  = nn.Parameter(torch.zeros(k))
        self.avg_user_lin = nn.Parameter(torch.zeros(1))
        self.avg_biz_lin  = nn.Parameter(torch.zeros(1))

        # --- KG FIELD PROCESSING ---
        self.use_kg = (user_kg_mat is not None and biz_kg_mat is not None)
        if self.use_kg:
            kg_dim = user_kg_mat.shape[1]

            # Frozen pretrained embeddings (lookup table, không train)
            self.user_kg_emb = nn.Embedding.from_pretrained(
                torch.from_numpy(user_kg_mat.copy()), freeze=True)
            self.biz_kg_emb  = nn.Embedding.from_pretrained(
                torch.from_numpy(biz_kg_mat.copy()), freeze=True)

            # Precompute KG presence mask (True = có KG, False = zero-vector)
            user_has_kg = torch.from_numpy(
                (np.abs(user_kg_mat).sum(axis=1) > 1e-8).astype(np.float32))
            biz_has_kg  = torch.from_numpy(
                (np.abs(biz_kg_mat).sum(axis=1) > 1e-8).astype(np.float32))
            self.register_buffer("user_has_kg", user_has_kg)
            self.register_buffer("biz_has_kg",  biz_has_kg)

            n_user_with_kg = int(user_has_kg.sum().item())
            n_biz_with_kg  = int(biz_has_kg.sum().item())
            print(f"  KG mask: {n_user_with_kg:,}/{len(user_has_kg):,} users, "
                  f"{n_biz_with_kg:,}/{len(biz_has_kg):,} biz have KG embeddings")

            # Projection layers: KG space (kg_dim) → FM latent space (k)
            self.user_kg_proj = nn.Linear(kg_dim, k)
            self.biz_kg_proj  = nn.Linear(kg_dim, k)

            # First-order linear weights cho KG
            self.user_kg_lin  = nn.Linear(kg_dim, 1)
            self.biz_kg_lin   = nn.Linear(kg_dim, 1)

            # Cross&Compress Units
            # user: u_emb (k) ⊗ u_kg_proj (k) → refined u_emb + refined u_kg
            # biz:  b_emb (k) ⊗ b_kg_proj (k) → refined b_emb + refined b_kg
            self.cc_user = CrossCompressUnit(k, use_matrix=use_matrix_cc)
            self.cc_biz  = CrossCompressUnit(k, use_matrix=use_matrix_cc)

            # Init projection & linear layers — đồng nhất std=0.01
            nn.init.normal_(self.user_kg_proj.weight, 0, 0.01)
            nn.init.normal_(self.biz_kg_proj.weight,  0, 0.01)
            nn.init.zeros_(self.user_kg_proj.bias)
            nn.init.zeros_(self.biz_kg_proj.bias)
            nn.init.normal_(self.user_kg_lin.weight, 0, 0.01)
            nn.init.normal_(self.biz_kg_lin.weight,  0, 0.01)
            nn.init.zeros_(self.user_kg_lin.bias)
            nn.init.zeros_(self.biz_kg_lin.bias)

        # --- STANDARD INIT ---
        for p in [self.user_lin.weight, self.biz_lin.weight,
                  self.user_emb.weight, self.biz_emb.weight]:
            nn.init.normal_(p, 0, 0.01)
        if self.num_lin is not None: nn.init.normal_(self.num_lin, 0, 0.01)
        if self.num_emb is not None: nn.init.normal_(self.num_emb, 0, 0.01)
        nn.init.normal_(self.avg_user_emb, 0, 0.01)
        nn.init.normal_(self.avg_biz_emb, 0, 0.01)

    def forward(self, user, biz, num, cold_user_mask=None, cold_biz_mask=None):
        # ===== FIRST-ORDER =====
        u_lin = self.user_lin(user).squeeze(-1)
        b_lin = self.biz_lin(biz).squeeze(-1)
        if cold_user_mask is not None:
            u_lin = torch.where(cold_user_mask, self.avg_user_lin.expand_as(u_lin), u_lin)
        if cold_biz_mask is not None:
            b_lin = torch.where(cold_biz_mask, self.avg_biz_lin.expand_as(b_lin), b_lin)

        linear = self.bias + u_lin + b_lin
        if self.num_lin is not None and num.shape[1] > 0:
            linear = linear + (num * self.num_lin).sum(dim=1)

        # ===== SECOND-ORDER EMBEDDINGS =====
        u_emb = self.user_emb(user)
        b_emb = self.biz_emb(biz)
        if cold_user_mask is not None:
            u_emb = torch.where(cold_user_mask.unsqueeze(-1),
                                self.avg_user_emb.expand_as(u_emb), u_emb)
        if cold_biz_mask is not None:
            b_emb = torch.where(cold_biz_mask.unsqueeze(-1),
                                self.avg_biz_emb.expand_as(b_emb), b_emb)

        # ===== KG FIELD PROCESSING + CROSS&COMPRESS =====
        kg_emb_fields = []
        if self.use_kg:
            u_kg_raw = self.user_kg_emb(user)   # (B, kg_dim) — frozen lookup
            b_kg_raw = self.biz_kg_emb(biz)     # (B, kg_dim) — frozen lookup

            # Zero-mask: (B, 1) — 1.0 if has KG, 0.0 if not
            u_kg_mask = self.user_has_kg[user].unsqueeze(-1)
            b_kg_mask = self.biz_has_kg[biz].unsqueeze(-1)

            # First-order KG contribution (masked)
            u_kg_lin_out = self.user_kg_lin(u_kg_raw).squeeze(-1) * u_kg_mask.squeeze(-1)
            b_kg_lin_out = self.biz_kg_lin(b_kg_raw).squeeze(-1) * b_kg_mask.squeeze(-1)
            linear = linear + u_kg_lin_out + b_kg_lin_out

            # Project KG → FM latent space, then mask
            u_kg_proj = self.user_kg_proj(u_kg_raw) * u_kg_mask  # (B, k)
            b_kg_proj = self.biz_kg_proj(b_kg_raw)  * b_kg_mask  # (B, k)

            # ── Cross&Compress ──────────────────────────────────────
            # Input:  CF embedding (u_emb/b_emb) ⊗ KG projection (u_kg_proj/b_kg_proj)
            # Output: refined embeddings carrying both CF and KG signals
            #
            # CRITICAL: CC outputs MUST be masked to prevent bias leak
            # When kg_mask=0, CC input u_kg_proj=0 → C=0 → output=bias≠0
            # Without masking, bias would inject noise for entities without KG
            u_emb_cc, u_kg_cc = self.cc_user(u_emb, u_kg_proj)  # (B,k), (B,k)
            b_emb_cc, b_kg_cc = self.cc_biz(b_emb,  b_kg_proj)  # (B,k), (B,k)

            # [FIX] Mask CC outputs — triệt tiêu bias leak cho entity không có KG
            u_emb_cc = u_emb_cc * u_kg_mask   # (B, k) — zero khi không có KG
            u_kg_cc  = u_kg_cc  * u_kg_mask   # (B, k)
            b_emb_cc = b_emb_cc * b_kg_mask   # (B, k)
            b_kg_cc  = b_kg_cc  * b_kg_mask   # (B, k)

            # Residual connection: giữ original CF signal, cộng thêm CC refinement
            # Khi mask=0: u_emb giữ nguyên, không bị nhiễu
            u_emb = u_emb + u_emb_cc
            b_emb = b_emb + b_emb_cc

            # KG fields với residual (proj + CC refined), masked
            kg_emb_fields = [
                (u_kg_proj + u_kg_cc).unsqueeze(1),  # (B, 1, k)
                (b_kg_proj + b_kg_cc).unsqueeze(1),  # (B, 1, k)
            ]
            # ────────────────────────────────────────────────────────

        # Build emb_list — u_emb/b_emb đã được CC refine (nếu KG enabled)
        emb_list = [u_emb.unsqueeze(1), b_emb.unsqueeze(1)]  # (B,1,k) each
        if self.num_emb is not None and num.shape[1] > 0:
            emb_list.append(num.unsqueeze(-1) * self.num_emb)  # (B, num_dim, k)
        emb_list.extend(kg_emb_fields)

        # ===== FM INTERACTION =====
        all_emb = self.dropout(torch.cat(emb_list, dim=1))  # (B, F, k)
        sum_sq = all_emb.sum(dim=1) ** 2         # (B, k)
        sq_sum = (all_emb ** 2).sum(dim=1)       # (B, k)
        inter  = 0.5 * (sum_sq - sq_sum).sum(dim=1)  # (B,)

        out = linear + inter
        if not self.training:
            out = torch.clamp(out, 0.0, 1.0)
        return out

# METRICS
# =========================================================
def compute_metrics(predictions, targets):
    p = torch.as_tensor(predictions).cpu().float()
    t = torch.as_tensor(targets).cpu().float()
    mse  = torch.mean((p-t)**2).item()
    rmse = math.sqrt(mse)
    mae  = torch.mean(torch.abs(p-t)).item()
    ss_res = torch.sum((t-p)**2).item()
    ss_tot = torch.sum((t-t.mean())**2).item()
    r2 = 1 - ss_res/ss_tot if ss_tot > 0 else 0.0
    return {"MSE":mse,"RMSE":rmse,"MAE":mae,"R2":r2}

def evaluate_model(model, loader, device):
    was_training = model.training
    model.eval()
    all_preds, all_targets = [], []
    with torch.no_grad():
        for user, biz, num, rating, cu, cb in loader:
            pred = model(user.to(device), biz.to(device), num.to(device),
                         cold_user_mask=cu.bool().to(device),
                         cold_biz_mask=cb.bool().to(device))
            all_preds.append(denormalize_target(pred.cpu()))
            all_targets.append(denormalize_target(rating))
    if was_training:
        model.train()
    return compute_metrics(torch.cat(all_preds), torch.cat(all_targets))

def print_metrics(m, prefix=""):
    print(f"{prefix}MSE:{m['MSE']:.4f} | RMSE:{m['RMSE']:.4f} | MAE:{m['MAE']:.4f} | R2:{m['R2']:.4f}")

def save_checkpoint(path, model, optimizer, scheduler, scaler, epoch,
                    best_val_rmse, best_epoch, history, no_improve):
    raw_model = model._orig_mod if hasattr(model, "_orig_mod") else model
    torch.save({
        "epoch":          epoch,
        "model_state":    raw_model.state_dict(),
        "optimizer_state":optimizer.state_dict(),
        "scheduler_state":scheduler.state_dict(),
        "scaler_state":   scaler.state_dict(),
        "best_val_rmse":  best_val_rmse,
        "best_epoch":     best_epoch,
        "history":        history,
        "no_improve":     no_improve,
    }, path)
    print(f"  💾 Checkpoint saved → {path}")


def load_checkpoint(path, model, optimizer, scheduler, scaler):
    ckpt = torch.load(path, weights_only=False)
    raw_model = model._orig_mod if hasattr(model, "_orig_mod") else model
    raw_model.load_state_dict(ckpt["model_state"])
    optimizer.load_state_dict(ckpt["optimizer_state"])
    scheduler.load_state_dict(ckpt["scheduler_state"])
    scaler.load_state_dict(ckpt["scaler_state"])
    print(f"  ✅ Checkpoint loaded from epoch {ckpt['epoch']} → {path}")
    return (ckpt["epoch"], ckpt["best_val_rmse"],
            ckpt["best_epoch"], ckpt["history"], ckpt["no_improve"])

def smooth_labels(ratings, alpha=0.05):
    return ratings * (1 - alpha) + 0.5 * alpha

def train_fm(all_data, n_user=None, n_business=None,
             user_kg_mat=None, biz_kg_mat=None,
             cold_threshold=COLD_START_THRESHOLD, epochs=300,
             batch_size=2048, lr=0.005, train_split=0.8, val_split=0.1,
             early_stopping_patience=30, early_stopping_min_delta=1e-4,
             eval_every=1, experiment_name="FM",
             checkpoint_dir="/kaggle/working", resume_from=None,
             use_matrix_cc=True):

    t_start = time.time()
    device  = "cuda" if torch.cuda.is_available() else "cpu"
    # if device == "cuda": torch.backends.cudnn.benchmark = True # Disabled for deterministic seed

    num_feat = all_data.tensors[2].shape[1]
    kg_status = "YES (field-level projection)" if (user_kg_mat is not None) else "NO"
    print(f"\n{'='*60}")
    print(f"[{experiment_name}] {num_feat} context features | KG: {kg_status}")
    print(f"{'='*60}")
    print(f"Device: {device}")

    n = len(all_data)
    set_seed(42)
    idx = torch.randperm(n)
    train_end = int(n * train_split)
    val_end   = train_end + int(n * val_split)
    tensors   = [t[idx] for t in all_data.tensors]

    train_data = TensorDataset(*[t[:train_end]        for t in tensors])
    val_data   = TensorDataset(*[t[train_end:val_end] for t in tensors])
    test_data  = TensorDataset(*[t[val_end:]          for t in tensors])
    print(f"Train:{len(train_data):,} | Val:{len(val_data):,} | Test:{len(test_data):,}")

    tu = train_data.tensors[0].numpy(); tb = train_data.tensors[1].numpy()
    max_u = int(all_data.tensors[0].max())
    max_b = int(all_data.tensors[1].max())
    u_cnt = np.bincount(tu, minlength=max_u+1)
    b_cnt = np.bincount(tb, minlength=max_b+1)

    train_cold_users  = np.sum((u_cnt <= cold_threshold) & (u_cnt > 0))
    train_cold_bizes  = np.sum((b_cnt <= cold_threshold) & (b_cnt > 0))
    train_active_users= np.sum(u_cnt > 0)
    train_active_bizes= np.sum(b_cnt > 0)
    print(f"Train set cold statistics (leak-free):")
    print(f"  Active users: {train_active_users:,} | Cold start (≤{cold_threshold}): {train_cold_users:,} "
          f"({train_cold_users/max(train_active_users,1)*100:.1f}%)")
    print(f"  Active biz  : {train_active_bizes:,} | Cold start (≤{cold_threshold}): {train_cold_bizes:,} "
          f"({train_cold_bizes/max(train_active_bizes,1)*100:.1f}%)")

    for ds in [train_data, val_data, test_data]:
        u_arr = ds.tensors[0].numpy(); b_arr = ds.tensors[1].numpy()
        ds.tensors[4].copy_(torch.from_numpy((u_cnt[u_arr] <= cold_threshold).astype(np.float32)))
        ds.tensors[5].copy_(torch.from_numpy((b_cnt[b_arr] <= cold_threshold).astype(np.float32)))

    class FastLoader:
        def __init__(self, ds, bs, shuffle=False):
            self.tensors = ds.tensors; self.bs = bs; self.shuffle = shuffle; self.n = ds.tensors[0].size(0)
        def __iter__(self):
            self.indices = torch.randperm(self.n) if self.shuffle else torch.arange(self.n)
            self.i = 0; return self
        def __next__(self):
            if self.i >= self.n: raise StopIteration
            idx = self.indices[self.i:self.i+self.bs]; self.i += self.bs
            return tuple(t[idx] for t in self.tensors)
        def __len__(self): return math.ceil(self.n / self.bs)

    try:
        if device != "cuda": raise RuntimeError("CPU")
        train_data = TensorDataset(*[t.to(device) for t in train_data.tensors])
        val_data   = TensorDataset(*[t.to(device) for t in val_data.tensors])
        test_data  = TensorDataset(*[t.to(device) for t in test_data.tensors])
        train_loader = FastLoader(train_data, batch_size, shuffle=True)
        val_loader   = FastLoader(val_data,   batch_size*2)
        test_loader  = FastLoader(test_data,  batch_size*2)
        print("Data loaded to GPU VRAM.")
    except RuntimeError:
        nw = min(4, os.cpu_count() or 1)
        train_loader = DataLoader(train_data, batch_size, shuffle=True,  pin_memory=True, num_workers=nw, persistent_workers=True, prefetch_factor=4, drop_last=True)
        val_loader   = DataLoader(val_data,   batch_size*2, pin_memory=True, num_workers=nw, persistent_workers=True, prefetch_factor=4)
        test_loader  = DataLoader(test_data,  batch_size*2, pin_memory=True, num_workers=nw, persistent_workers=True, prefetch_factor=4)

    if n_user     is None: n_user     = int(all_data.tensors[0].max())+1
    if n_business is None: n_business = int(all_data.tensors[1].max())+1

    model = FMModel(n_user, n_business, num_dim=num_feat, k=64, dropout=0.15,
                    user_kg_mat=user_kg_mat, biz_kg_mat=biz_kg_mat, use_matrix_cc=use_matrix_cc).to(device)

    total_params    = sum(p.numel() for p in model.parameters())
    trainable_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
    frozen_params    = total_params - trainable_params
    print(f"Params total: {total_params:,} | Trainable: {trainable_params:,} | Frozen (KG): {frozen_params:,}")

    if hasattr(torch,"compile"):
        try: model = torch.compile(model, mode="reduce-overhead")
        except: pass

    optimizer = optim.SGD(model.parameters(), lr=lr, momentum=0.9, weight_decay=1e-5, nesterov=True)
    loss_fn   = nn.HuberLoss(delta=0.5)
    warmup    = optim.lr_scheduler.LinearLR(optimizer, start_factor=1/WARMUP_EPOCHS, total_iters=WARMUP_EPOCHS)
    cosine    = optim.lr_scheduler.CosineAnnealingWarmRestarts(optimizer, T_0=30, T_mult=2, eta_min=1e-5)
    scheduler = optim.lr_scheduler.SequentialLR(optimizer, schedulers=[warmup, cosine], milestones=[WARMUP_EPOCHS])
    scaler    = torch.amp.GradScaler('cuda', enabled=(device=="cuda"))

    best_val_rmse = float("inf"); best_state = None; best_epoch = 0; no_improve = 0
    history = {"train_loss":[],"val_rmse":[],"val_mae":[],"val_r2":[]}
    start_epoch = 0

    ckpt_path = os.path.join(checkpoint_dir, f"{experiment_name}_checkpoint.pt")
    if resume_from is not None:
        resume_path = resume_from if isinstance(resume_from, str) else ckpt_path
        if os.path.exists(resume_path):
            start_epoch, best_val_rmse, best_epoch, history, no_improve = \
                load_checkpoint(resume_path, model, optimizer, scheduler, scaler)
            raw_model = model._orig_mod if hasattr(model, "_orig_mod") else model
            best_state = copy.deepcopy(raw_model.state_dict())
            print(f"  Resuming from epoch {start_epoch+1}, best RMSE: {best_val_rmse:.4f}")
        else:
            print(f"  ⚠️  Checkpoint not found at {resume_path}, starting fresh.")

    print(f"\nTraining [{experiment_name}]...\n")
    for ep in range(start_epoch, epochs):
        model.train()
        total_loss = total_count = 0
        for user, biz, num, rating, cu, cb in train_loader:
            user   = user.to(device, non_blocking=True)
            biz    = biz.to(device, non_blocking=True)
            num    = num.to(device, non_blocking=True)
            rating = rating.to(device, non_blocking=True)
            optimizer.zero_grad()
            with torch.amp.autocast('cuda', enabled=(device=="cuda")):
                pred = model(user, biz, num,
                             cold_user_mask=cu.bool().to(device, non_blocking=True),
                             cold_biz_mask=cb.bool().to(device, non_blocking=True))
                smooth_rating = smooth_labels(rating, alpha=0.05)
                loss = loss_fn(pred, smooth_rating)
            scaler.scale(loss).backward()
            scaler.unscale_(optimizer)
            torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
            scaler.step(optimizer); scaler.update()
            total_loss += loss.item()*len(user); total_count += len(user)

        avg_loss = total_loss/total_count if total_count > 0 else float("nan")
        history["train_loss"].append(avg_loss)
        scheduler.step()

        if (ep+1) % eval_every == 0:
            vm = evaluate_model(model, val_loader, device)
            lr_now = optimizer.param_groups[0]["lr"]
            history["val_rmse"].append(vm["RMSE"]); history["val_mae"].append(vm["MAE"]); history["val_r2"].append(vm["R2"])
            print(f"Epoch {ep+1:03d}/{epochs} | Loss:{avg_loss:.6f} | LR:{lr_now:.6f}")
            print_metrics(vm, "  Val → ")

            if vm["RMSE"] < best_val_rmse - early_stopping_min_delta:
                best_val_rmse = vm["RMSE"]
                raw_model = model._orig_mod if hasattr(model, "_orig_mod") else model
                best_state = copy.deepcopy(raw_model.state_dict())
                best_epoch = ep+1; no_improve = 0
                print(f"  ★ Best RMSE: {best_val_rmse:.4f}")
                os.makedirs(checkpoint_dir, exist_ok=True)
                save_checkpoint(ckpt_path, model, optimizer, scheduler, scaler,
                                ep, best_val_rmse, best_epoch, history, no_improve)
            else:
                no_improve += 1
                print(f"  No improve {no_improve}/{early_stopping_patience}")
                if no_improve >= early_stopping_patience:
                    print(f"\nEarly stopping at epoch {ep+1}")
                    history["stopped_epoch"] = ep+1; break

    if best_state:
        raw_model = model._orig_mod if hasattr(model, "_orig_mod") else model
        raw_model.load_state_dict(best_state)
        print(f"\nLoaded best model (epoch {best_epoch}, Val RMSE:{best_val_rmse:.4f})")

    train_time = time.time() - t_start
    print(f"\n{'='*60}\nFINAL EVALUATION [{experiment_name}]\n{'='*60}")
    results = {}
    for split, loader in [("Train",train_loader),("Val",val_loader),("Test",test_loader)]:
        m = evaluate_model(model, loader, device)
        print(f"\n{split}:"); print_metrics(m, "  ")
        results[split] = m

    history.update({"best_epoch":best_epoch,"best_val_rmse":best_val_rmse,
                    "train_time_sec":train_time,"num_features":num_feat,
                    "experiment_name":experiment_name,
                    "kg_enabled": user_kg_mat is not None})
    print(f"\n  Train time: {train_time/60:.1f} min ({train_time:.0f}s)")
    return model, history, test_data, test_loader, train_data, val_data

def save_outputs(results, out_dir="/kaggle/working"):
    os.makedirs(out_dir, exist_ok=True)

    fm_path = os.path.join(out_dir, "fm_model.pt")
    raw_model = results["fm_model"]
    if hasattr(raw_model, "_orig_mod"):
        raw_model = raw_model._orig_mod
    torch.save(raw_model.state_dict(), fm_path)
    print(f"  ✓ FM model       → {fm_path}")

    id_map_path = os.path.join(out_dir, "id_maps.pkl")
    with open(id_map_path, "wb") as f:
        pickle.dump({"user_map":results["user_map"].map,"biz_map":results["biz_map"].map}, f)

    history_path = os.path.join(out_dir, "fm_history.json")
    history_s = {k:([float(x) for x in v] if isinstance(v,list) else v)
                 for k,v in results["fm_history"].items()}
    with open(history_path, "w") as f:
        json.dump(history_s, f, indent=2)
    print(f"  ✓ FM history     → {history_path}")

    info_path = os.path.join(out_dir, "model_info.json")
    info = {
        "num_features_used":  results.get("num_features_used", NUM_DIM),
        "n_users":            len(results["user_map"].map),
        "n_businesses":       len(results["biz_map"].map),

        "best_val_rmse":      results["fm_history"].get("best_val_rmse"),
        "best_epoch":         results["fm_history"].get("best_epoch"),
        "stopped_epoch":      results["fm_history"].get("stopped_epoch", None),
        "train_time_sec":     results["fm_history"].get("train_time_sec"),
        "train_time_min":     results["fm_history"].get("train_time_sec",0)/60,

        "cold_threshold":     COLD_START_THRESHOLD,

    print(f"\n  All outputs saved to: {out_dir}/")

def run_full_pipeline(review_path, user_ctx, biz_ctx, user_tip, biz_tip,
                      checkin_ctx, user_kg_emb=None, biz_kg_emb=None,
                      max_reviews=None, cold_threshold=COLD_START_THRESHOLD,
                      selected_features=None, checkpoint_dir="/kaggle/working",
                      resume_from=None, use_matrix_cc=True, **fm_kwargs):

    print("\n" + "="*60)
    print("LOADING ALL DATA INTO MEMORY...")
    print("="*60)
    all_data, user_map, biz_map, user_counts, biz_counts, user_kg_mat, biz_kg_mat = load_all_data(
        review_path=review_path,
        user_ctx=user_ctx, biz_ctx=biz_ctx,
        user_tip=user_tip, biz_tip=biz_tip,
        checkin_ctx=checkin_ctx,
        user_kg_emb=user_kg_emb, biz_kg_emb=biz_kg_emb,
        max_reviews=max_reviews, cold_threshold=cold_threshold,
        selected_features=selected_features)

    cold_users = sum(1 for c in user_counts.values() if c <= cold_threshold)
    cold_bizes = sum(1 for c in biz_counts.values() if c <= cold_threshold)
    num_features_active = all_data.tensors[2].shape[1]

    print(f"Total users: {len(user_counts):,} | Cold start (≤{cold_threshold}): {cold_users:,} "
          f"({cold_users/max(len(user_counts),1)*100:.1f}%)")
    print(f"Total biz:   {len(biz_counts):,} | Cold start (≤{cold_threshold}): {cold_bizes:,} "
          f"({cold_bizes/max(len(biz_counts),1)*100:.1f}%)")
    print(f"Total samples: {len(all_data):,}")
    print(f"Numerical feature dim: {num_features_active} (context only, KG handled separately)")
    if user_kg_mat is not None:
        print(f"KG user matrix: {user_kg_mat.shape} | KG biz matrix: {biz_kg_mat.shape}")
    else:
        print("KG embeddings: DISABLED")

    n_user     = len(user_map.map)
    n_business = len(biz_map.map)

    fm_model, fm_history, test_data, test_loader, train_data, val_data = train_fm(
        all_data, n_user=n_user, n_business=n_business,
        user_kg_mat=user_kg_mat, biz_kg_mat=biz_kg_mat,
        cold_threshold=cold_threshold,
        checkpoint_dir=checkpoint_dir, resume_from=resume_from,
        use_matrix_cc=use_matrix_cc,
        **fm_kwargs)

    return {
        "fm_model":          fm_model,
        "fm_history":        fm_history,
        "user_map":          user_map,
        "biz_map":           biz_map,
        "train_data":        train_data,
        "val_data":          val_data,
        "test_data":         test_data,
        "num_features_used": num_features_active,
        "selected_features": selected_features if selected_features is not None else FEAT_NAMES,
    }
